cmake_minimum_required(VERSION 3.25)
project(MasterRallyeRuntimeLauncher LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "R5V-J.1 launcher is Windows-only")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "R5V-J.1 launcher must be built as x64 to inspect a WOW64 target")
endif()
if(NOT PROFILE_INCLUDE_DIR)
  message(FATAL_ERROR "PROFILE_INCLUDE_DIR must point to the generated exact-build trust header")
endif()

add_executable(mr-runtime-launcher src/main.cpp)
target_include_directories(mr-runtime-launcher PRIVATE "${PROFILE_INCLUDE_DIR}")
target_compile_features(mr-runtime-launcher PRIVATE cxx_std_20)
target_compile_definitions(mr-runtime-launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(mr-runtime-launcher PRIVATE /W4 /permissive- /EHsc)
target_link_options(mr-runtime-launcher PRIVATE /Brepro)
target_link_libraries(mr-runtime-launcher PRIVATE bcrypt)
set_target_properties(mr-runtime-launcher PROPERTIES OUTPUT_NAME "mr-runtime-launcher")
