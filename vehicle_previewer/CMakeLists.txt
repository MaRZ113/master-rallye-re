cmake_minimum_required(VERSION 3.25)
project(MasterRallyeVehiclePreviewer LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Previewer currently requires Windows / Direct3D 9")
endif()
add_executable(mr-vehicle-previewer WIN32 src/main.cpp)
target_compile_features(mr-vehicle-previewer PRIVATE cxx_std_20)
target_compile_definitions(mr-vehicle-previewer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
target_compile_definitions(mr-vehicle-previewer PRIVATE SDK_ROOT=L"${SDK_ROOT}")
target_compile_options(mr-vehicle-previewer PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_options(mr-vehicle-previewer PRIVATE /Brepro)
target_link_libraries(mr-vehicle-previewer PRIVATE d3d9 comdlg32 shell32)
