cmake_minimum_required(VERSION 3.21)
project(MRRRenderer LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Configure a Windows x86/Win32 build")
endif()
if(NOT MSVC)
 message(FATAL_ERROR "This initial build uses MSVC x86, including guarded metadata reads")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CORE src/forwarders.cpp src/wrappers.cpp src/state_tracker.cpp src/trace.cpp src/provenance.cpp src/guids.cpp src/visual_policy.cpp src/visual_wrappers.cpp src/vehicle_classifier.cpp src/vehicle_semantics.cpp src/reflection_scope.cpp src/game_fov.cpp src/quality.cpp src/quality_wrappers.cpp src/compatibility.cpp src/menu_freeze.cpp src/ui_margins.cpp)
add_library(proxy_core STATIC ${CORE})
target_include_directories(proxy_core PUBLIC include vendor/d3d8)
target_compile_options(proxy_core PRIVATE /W4 /permissive- /fp:strict /EHsc /Zi)
target_link_libraries(proxy_core PUBLIC bcrypt user32)
add_library(d3d8 SHARED src/dllmain.cpp src/real_d3d8.cpp exports.def)
target_link_libraries(d3d8 PRIVATE proxy_core)
target_link_options(d3d8 PRIVATE /INCREMENTAL:NO /DEBUG /Brepro /MAP)
set_target_properties(d3d8 PROPERTIES PREFIX "" OUTPUT_NAME d3d8)
include(CTest)
